cmake_minimum_required(VERSION 3.22.1)
project(PortableUnityLibraries C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(NOT SOURCE_ROOT)
  message(FATAL_ERROR "SOURCE_ROOT must contain the hash-verified sources")
endif()
if(ANDROID AND NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Only Android ARM64 is supported")
endif()
function(portable_library target)
  add_library(${target} SHARED ${ARGN})
  target_compile_options(${target} PRIVATE -O2 -fstack-protector-strong)
  target_link_options(${target} PRIVATE -Wl,--no-undefined -Wl,-Bsymbolic-functions -Wl,-z,relro -Wl,-z,now)
  if(ANDROID)
    target_link_options(${target} PRIVATE -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
  endif()
endfunction()

portable_library(sqlite3 "${SOURCE_ROOT}/sqlite/sqlite3.c")
target_include_directories(sqlite3 PUBLIC "${SOURCE_ROOT}/sqlite")
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_ENABLE_COLUMN_METADATA=1 SQLITE_ENABLE_FTS5=1 SQLITE_ENABLE_RTREE=1)
target_link_libraries(sqlite3 PRIVATE m ${CMAKE_DL_LIBS})

set(LZ4 "${SOURCE_ROOT}/lz4/lib")
portable_library(lz4 "${LZ4}/lz4.c" "${LZ4}/lz4hc.c" "${LZ4}/lz4frame.c" "${LZ4}/xxhash.c")
target_include_directories(lz4 PUBLIC "${LZ4}")
target_link_options(lz4 PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/lz4.exports")

set(ZSTD "${SOURCE_ROOT}/zstd/lib")
file(GLOB ZSTD_SOURCES CONFIGURE_DEPENDS "${ZSTD}/common/*.c" "${ZSTD}/compress/*.c" "${ZSTD}/decompress/*.c" "${ZSTD}/dictBuilder/*.c" "${ZSTD}/legacy/*.c")
portable_library(zstd ${ZSTD_SOURCES})
target_include_directories(zstd PUBLIC "${ZSTD}" PRIVATE "${ZSTD}/common")
target_compile_definitions(zstd PRIVATE ZSTD_MULTITHREAD=1 ZSTD_LEGACY_SUPPORT=1 ZSTD_DISABLE_ASM=1)
find_package(Threads REQUIRED)
target_link_libraries(zstd PRIVATE Threads::Threads)

set(OGG "${SOURCE_ROOT}/ogg")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/include/ogg")
file(WRITE "${CMAKE_BINARY_DIR}/include/ogg/config_types.h" "#include <stdint.h>\ntypedef int16_t ogg_int16_t;\ntypedef uint16_t ogg_uint16_t;\ntypedef int32_t ogg_int32_t;\ntypedef uint32_t ogg_uint32_t;\ntypedef int64_t ogg_int64_t;\ntypedef uint64_t ogg_uint64_t;\n")
portable_library(ogg "${OGG}/src/bitwise.c" "${OGG}/src/framing.c")
target_include_directories(ogg PUBLIC "${OGG}/include" "${CMAKE_BINARY_DIR}/include")

set(VORBIS "${SOURCE_ROOT}/vorbis")
set(VORBIS_FILES mdct smallft block envelope window lsp lpc analysis synthesis psy info floor1 floor0 res0 mapping0 registry codebook sharedbook lookup bitrate)
set(VORBIS_SOURCES)
foreach(name IN LISTS VORBIS_FILES)
  list(APPEND VORBIS_SOURCES "${VORBIS}/lib/${name}.c")
endforeach()
portable_library(vorbis ${VORBIS_SOURCES})
target_include_directories(vorbis PUBLIC "${VORBIS}/include" PRIVATE "${VORBIS}/lib")
target_link_libraries(vorbis PUBLIC ogg PRIVATE m)
portable_library(vorbisfile "${VORBIS}/lib/vorbisfile.c")
target_include_directories(vorbisfile PUBLIC "${VORBIS}/include")
target_link_libraries(vorbisfile PUBLIC vorbis ogg PRIVATE m)
portable_library(vorbisenc "${VORBIS}/lib/vorbisenc.c")
target_include_directories(vorbisenc PUBLIC "${VORBIS}/include" PRIVATE "${VORBIS}/lib")
target_link_libraries(vorbisenc PUBLIC vorbis ogg PRIVATE m)

set(PNG "${SOURCE_ROOT}/png")
configure_file("${PNG}/scripts/pnglibconf.h.prebuilt" "${CMAKE_BINARY_DIR}/include/pnglibconf.h" COPYONLY)
set(PNG_FILES png pngerror pngget pngmem pngpread pngread pngrio pngrtran pngrutil pngset pngtrans pngwio pngwrite pngwtran pngwutil)
set(PNG_SOURCES)
foreach(name IN LISTS PNG_FILES)
  list(APPEND PNG_SOURCES "${PNG}/${name}.c")
endforeach()
portable_library(png16 ${PNG_SOURCES})
target_include_directories(png16 PUBLIC "${PNG}" "${CMAKE_BINARY_DIR}/include")
target_compile_definitions(png16 PRIVATE PNG_ARM_NEON_OPT=0)
find_package(ZLIB REQUIRED)
target_link_libraries(png16 PRIVATE ZLIB::ZLIB m)

option(BUILD_TESTING "Run functional tests on the build host" OFF)
if(BUILD_TESTING AND NOT ANDROID)
  enable_testing()
  add_executable(native_provider_smoke tests/smoke.c)
  target_link_libraries(native_provider_smoke sqlite3 lz4 zstd ogg vorbis vorbisfile vorbisenc png16 m)
  foreach(name sqlite lz4 zstd ogg vorbis vorbisenc vorbisfile png16)
    add_test(NAME ${name} COMMAND native_provider_smoke ${name})
  endforeach()
endif()
